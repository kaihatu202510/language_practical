(11)
select
  name as "製品名",
  sum(quantity) as "注文合計数"
from
  order_item
  join
    item on order_item.item_id = item.item_id
group by
  name
order by
  注文合計数 desc;


(12)
select
  order_id as "注文ID",
  sum(price * quantity) as "合計金額"
from
  order_item
  join
    item on order_item.item_id = item.item_id
group by
  order_id
order by
  合計金額 desc;


(13)
select account.account_id as ユーザーID, account.name as ユーザー名, sum(quantity * price) as 合計金額
from account
join order_info on account.account_id = order_info.account_id
join order_item on order_info.order_info_id = order_item.order_id
join item on order_item.item_id = item.item_id
group by account.account_id
order by account.account_id;
